//init
cd ~/ros2_ws
colcon build --symlink-install
source /opt/ros/jazzy/setup.bash
source ~/ros2_ws/install/setup.bash

//run server
ros2 launch crazyflie launch.py \
  crazyflies_yaml_file:=$HOME/ros2_ws/src/touch-vision-drone-control/config/crazyflies.yaml \
  teleop:=false \
  gui:=false

//after running server u can run test flights, change the height(in meters) if ever
//takeoff
ros2 service call /cf1/takeoff crazyflie_interfaces/srv/Takeoff "{height: 0.6, duration: {sec: 2}}"
//land
ros2 service call /cf1/land crazyflie_interfaces/srv/Land "{height: 0.0, duration: {sec: 2}}"

//nodes = active scripts
//topic = communication bus for exchanging of data (prng router)
ros2 node list
ros2 topic list

//check dongle
lsusb | grep -i bitcraze

//bring out handy gui to see code connections
rqt_graph
